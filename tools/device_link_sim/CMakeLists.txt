cmake_minimum_required(VERSION 3.22)
project(NcsDeviceLinkSim LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core)
include(CTest)
add_library(device_link_core STATIC core/message_frame.cpp core/message_frame.h)
target_link_libraries(device_link_core PUBLIC Qt6::Core)
target_include_directories(device_link_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(BUILD_TESTING)
  find_package(Qt6 6.2 REQUIRED COMPONENTS Test)
  add_executable(device_link_core_test tests/message_frame_test.cpp)
  target_link_libraries(device_link_core_test PRIVATE device_link_core Qt6::Test)
  add_test(NAME device_link_core_test COMMAND device_link_core_test)
endif()
